1. Get all orders placed on a specific date

SELECT *
FROM ORDERS
WHERE O_ORDERDATE = '2024-01-01';


2. Find the top 5 most expensive orders
SELECT *
FROM ORDERS
ORDER BY O_TOTALPRICE DESC
LIMIT 5;


3. Count the number of orders per order status

SELECT O_ORDERSTATUS, COUNT(*) AS order_count
FROM ORDERS
GROUP BY O_ORDERSTATUS;


4. Average order price per order priority

SELECT O_ORDERPRIORITY, AVG(O_TOTALPRICE) AS avg_price
FROM ORDERS
GROUP BY O_ORDERPRIORITY;


5. Get orders where the comment mentions 'urgent'

SELECT *
FROM ORDERS
WHERE LOWER(O_COMMENT) LIKE '%urgent%';


6. Total sales per day (useful for trending)

SELECT O_ORDERDATE, SUM(O_TOTALPRICE) AS daily_sales
FROM ORDERS
GROUP BY O_ORDERDATE
ORDER BY O_ORDERDATE;


8. Retrieve all orders handled by a specific clerk

SELECT *

9. List order priorities in order of frequency

SELECT O_ORDERPRIORITY, COUNT(*) AS count
FROM ORDERS
GROUP BY O_ORDERPRIORITY
ORDER BY count DESC;

34. List all orders and show whether they are ‘High Value’ (above 100,000) or not


SELECT *,
       CASE 
           WHEN O_TOTALPRICE > 100000 THEN 'High Value'
           ELSE 'Normal'
       END AS order_categoryFROM ORDERS;

35. Show cumulative sales per customer (running total)


SELECT O_CUSTKEY, 
       O_ORDERDATE,
       O_TOTALPRICE,
       SUM(O_TOTALPRICE) OVER (PARTITION BY O_CUSTKEY ORDER BY O_ORDERDATE) AS cumulative_spendFROM ORDERS;

36. Find the number of orders each clerk processed per month


SELECT O_CLERK,
       DATE_TRUNC('MONTH', O_ORDERDATE) AS order_month,
       COUNT(*) AS orders_handledFROM ORDERSGROUP BY O_CLERK, order_monthORDER BY O_CLERK, order_month;

37. Get top 2 orders by total price per customer


SELECT *FROM (
    SELECT *,
           ROW_NUMBER() OVER (PARTITION BY O_CUSTKEY ORDER BY O_TOTALPRICE DESC) AS rn
    FROM ORDERS
)WHERE rn <= 2;

38. Flag orders placed more than 5 years ago


SELECT *,
       CASE 
           WHEN O_ORDERDATE < DATEADD(YEAR, -5, CURRENT_DATE) THEN 'Old'
           ELSE 'Recent'
       END AS order_age_flagFROM ORDERS;
